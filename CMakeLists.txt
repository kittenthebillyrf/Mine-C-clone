cmake_minimum_required(VERSION 3.20)
project(MineCppClone VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(MineCppClone src/main.cpp)
if(MSVC)
  target_compile_options(MineCppClone PRIVATE /W4 /permissive-)
else()
  target_compile_options(MineCppClone PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_test(NAME terrain_smoke_test COMMAND MineCppClone --size 32 --seed 1122 --output terrain_test.ppm)
