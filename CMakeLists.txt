cmake_minimum_required(VERSION 3.20)
project(MineCppClone VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(MineCppClone
  src/main.cpp
  src/i18n/Localization.cpp
)
if(MSVC)
  target_compile_options(MineCppClone PRIVATE /W4 /permissive- /utf-8)
else()
  target_compile_options(MineCppClone PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_test(NAME terrain_smoke_test COMMAND MineCppClone --size 32 --seed 1122 --output terrain_test.ppm --lang en)
add_test(NAME localization_russian COMMAND MineCppClone --size 16 --seed 1122 --output terrain_ru.ppm --lang ru)
add_test(NAME localization_spanish COMMAND MineCppClone --size 16 --seed 1122 --output terrain_es.ppm --lang es)
add_test(NAME localization_german COMMAND MineCppClone --size 16 --seed 1122 --output terrain_de.ppm --lang de)
add_test(NAME localization_french COMMAND MineCppClone --size 16 --seed 1122 --output terrain_fr.ppm --lang fr)
add_test(NAME localization_chinese COMMAND MineCppClone --size 16 --seed 1122 --output terrain_zh.ppm --lang zh_CN)
